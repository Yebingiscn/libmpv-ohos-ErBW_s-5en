cmake_minimum_required(VERSION 3.20)
project(ThinLTOAbiRegression LANGUAGES C CXX)
include(CheckTypeSize)
set(CMAKE_EXTRA_INCLUDE_FILES sys/types.h)
check_type_size("ssize_t" ABI_SSIZE_T LANGUAGE C)
file(WRITE "${CMAKE_BINARY_DIR}/abi.txt"
    "C_BYTE_ORDER=${CMAKE_C_BYTE_ORDER}\nCXX_BYTE_ORDER=${CMAKE_CXX_BYTE_ORDER}\n"
    "C_ABI=${CMAKE_C_COMPILER_ABI}\nCXX_ABI=${CMAKE_CXX_COMPILER_ABI}\n"
    "C_PTR=${CMAKE_C_SIZEOF_DATA_PTR}\nCXX_PTR=${CMAKE_CXX_SIZEOF_DATA_PTR}\nSSIZE_T=${ABI_SSIZE_T}\n")
if(REQUIRE_ABI)
    if(NOT CMAKE_C_BYTE_ORDER STREQUAL "LITTLE_ENDIAN" OR
       NOT CMAKE_CXX_BYTE_ORDER STREQUAL "LITTLE_ENDIAN" OR
       NOT CMAKE_C_COMPILER_ABI STREQUAL "ELF" OR
       NOT CMAKE_CXX_COMPILER_ABI STREQUAL "ELF" OR
       NOT CMAKE_C_SIZEOF_DATA_PTR EQUAL 8 OR
       NOT CMAKE_CXX_SIZEOF_DATA_PTR EQUAL 8 OR NOT ABI_SSIZE_T EQUAL 8)
        message(FATAL_ERROR "ThinLTO ABI/type detection failed")
    endif()
endif()
add_library(abi_bitcode STATIC entry.c)
